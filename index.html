2. PROSEDUR DEPLOYMENT SPREADSHEET:

  1. Buat Google Sheets baru (misal: DB_Hidroponik_Syuhada).
  2. Isi baris ke-1 (Kolom A1 s/d I1) dengan header:
    Waktu | Sesi | TDS | pH | Suhu | Validitas | Status_Kontrol | Durasi_Pompa_A | Durasi_Pompa_B
  3. Buka Ekstensi (Extensions) → Apps Script, lalu tempel kode berikut:
Code.gs 
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    data.waktu || new Date().toLocaleString("id-ID"),
    data.sesi || 1,
    data.tds,
    data.ph,
    data.suhu,
    data.validitas || "VALID",
    data.status_kontrol,
    data.durasi_a || 0,
    data.durasi_b || 0
  ]);
  return ContentService.createTextOutput(JSON.stringify({"status": "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var result = [];
  for (var i = 1; i < rows.length; i++) {
    result.push({
      waktu: rows[i][0], sesi: rows[i][1], tds: rows[i][2],
      ph: rows[i][3], suhu: rows[i][4], validitas: rows[i][5],
      status_kontrol: rows[i][6], durasi_a: rows[i][7], durasi_b: rows[i][8]
    });
  }
  return ContentService.createTextOutput(JSON.stringify(result.slice(-30).reverse()))
    .setMimeType(ContentService.MimeType.JSON);
}
